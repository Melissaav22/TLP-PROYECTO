# ------------------------------------------------------------
# Gramatica LL(1) unificada: MiniC (subconjunto de C) + espanol
# Formato:   NoTerminal -> alternativa1 | alternativa2
#   - 'eps' es epsilon (cadena vacia)
#   - Los terminales son nombres de tokens definidos en minic/lexer.py
#     y en minic/lexicon_es.py
#   - El simbolo inicial es el de la primera regla (Program)
# Referencia: sintaxis de C en el Anexo A de ISO C99/C11 y K&R, apendice A13.
# ------------------------------------------------------------

Program     -> StmtList | Oracion
StmtList    -> Stmt StmtList | eps
Stmt        -> Decl | Assign | If | While | For

# declaraciones y asignaciones
Decl        -> INT ID DeclTail SEMI
DeclTail    -> ASSIGN Expr | eps
Assign      -> AssignCore SEMI
AssignCore  -> ID ASSIGN Expr

# estructuras de control (el cuerpo siempre va entre llaves)
If          -> IF LPAREN Cond RPAREN Block ElsePart
ElsePart    -> ELSE ElseTail | eps
ElseTail    -> Block | If
While       -> WHILE LPAREN Cond RPAREN Block
For         -> FOR LPAREN ForInit SEMI Cond SEMI ForStep RPAREN Block
ForInit     -> INT ID DeclTail | AssignCore | eps
ForStep     -> AssignCore | eps
Block       -> LBRACE StmtList RBRACE

# condiciones
Cond        -> Expr RelOp Expr
RelOp       -> LT | GT | EQ | LE | GE | NE

# expresiones aritmeticas (sin recursion izquierda; * y / ligan mas que + y -)
Expr        -> Term ExprP
ExprP       -> PLUS Term ExprP | MINUS Term ExprP | eps
Term        -> Factor TermP
TermP       -> TIMES Factor TermP | DIVIDE Factor TermP | eps
Factor      -> LPAREN Expr RPAREN | ID | NUM

# ------------------------------------------------------------
# Extension al espanol
# La primera regla admite tanto Programas MiniC como Oraciones.
# Los rasgos morfologicos van en los nombres de los tokens y de los
# no terminales (MS/FS/MP/FP = masculino/femenino singular/plural).
# Para analizar una oracion hay que activar el lexico espanol con --es.
# ------------------------------------------------------------

Oracion     -> OracionSG | OracionPL | InvertidaSG | InvertidaPL
OracionSG   -> SN_MS SigueMS | SN_FS SigueFS
OracionPL   -> SN_MP SigueMP | SN_FP SigueFP

# Se lee el primer miembro antes de decidir entre sujeto simple y coordinado.
# La coordinacion exige plural. Solo la coordinacion enteramente femenina
# conserva femenino en el atributo; un miembro masculino selecciona SV_MP.
SigueMS     -> SV_MS | CONJ_Y CoordinadoM
SigueFS     -> SV_FS | CONJ_Y CoordinadoF
SigueMP     -> SV_MP | CONJ_Y CoordinadoM
SigueFP     -> SV_FP | CONJ_Y CoordinadoF
CoordinadoM -> SN ColaCoordM
CoordinadoF -> SN_MS ColaCoordM | SN_FS ColaCoordF | SN_MP ColaCoordM | SN_FP ColaCoordF
ColaCoordM  -> CONJ_Y CoordinadoM | SV_MP
ColaCoordF  -> CONJ_Y CoordinadoF | SV_FP

# Inversion acotada: verbo pleno + sujeto simple concordado + modificadores.
# No se admiten objeto adicional, sujeto coordinado ni copula en esta rama.
InvertidaSG -> VerboSG SujetoInvSG ModsS
InvertidaPL -> VerboPL SujetoInvPL ModsP
SujetoInvSG -> SN_MS | SN_FS
SujetoInvPL -> SN_MP | SN_FP

# sintagma nominal (con determinante obligatorio)
SN          -> SN_MS | SN_FS | SN_MP | SN_FP
SN_MS       -> DET_MS SUST_MS AdjMS
SN_FS       -> DET_FS NombreFS AdjFS
SN_MP       -> DET_MP SUST_MP AdjMP
SN_FP       -> DET_FP SUST_FP AdjFP
NombreFS    -> SUST_FS | SUST_MASA_FS

# adjetivos pospuestos
AdjMS       -> ADJ_MS AdjMS | eps
AdjFS       -> ADJ_FS AdjFS | eps
AdjMP       -> ADJ_MP AdjMP | eps
AdjFP       -> ADJ_FP AdjFP | eps

# frase preposicional
PP          -> PREP SN

# sintagma verbal: verbo pleno (+ subordinada o modificadores) o copula
# (+ atributo y modificadores). El verbo concuerda en numero con el sujeto.
SV_MS       -> VerboSG Cpl_MS | CopulaSG PredCpl_MS
SV_FS       -> VerboSG Cpl_FS | CopulaSG PredCpl_FS
SV_MP       -> VerboPL Cpl_MP | CopulaPL PredCpl_MP
SV_FP       -> VerboPL Cpl_FP | CopulaPL PredCpl_FP
VerboSG     -> VERB_3S | VERB_PAST_3S
VerboPL     -> VERB_3P | VERB_PAST_3P
CopulaSG    -> COP_3S | COP_PAST_3S
CopulaPL    -> COP_3P | COP_PAST_3P

# Complemento: subordinada o un objeto opcional seguido de modificadores.
# Solo los nombres de masa habilitados admiten objeto singular sin articulo.
Cpl_MS      -> QUE Oracion | Objeto ModsS
Cpl_FS      -> QUE Oracion | Objeto ModsS
Cpl_MP      -> QUE Oracion | Objeto ModsP
Cpl_FP      -> QUE Oracion | Objeto ModsP
Objeto      -> SN | MasaObjeto | eps
MasaObjeto  -> SUST_MASA_FS AdjFS

# complemento de la copula: atributo (adjetivo concordado) y modificadores
PredCpl_MS  -> PredMS ModsS | ModsS
PredCpl_FS  -> PredFS ModsS | ModsS
PredCpl_MP  -> PredMP ModsP | ModsP
PredCpl_FP  -> PredFP ModsP | ModsP

PredMS      -> ADJ_MS
PredFS      -> ADJ_FS
PredMP      -> ADJ_MP
PredFP      -> ADJ_FP

# modificadores del sintagma verbal (adverbios y frases preposicionales)
ModsS       -> ADV ModsS | PP ModsS | eps
ModsP       -> ADV ModsP | PP ModsP | eps
